from flask import Flask, render_template, request from Bio import SeqIO, pairwise2 from io import StringIO app = Flask(__name__) def parse_fasta(fasta_text): fasta_io = StringIO(fasta_text) records = list(SeqIO.parse(fasta_io, "fasta")) return records def calculate_gc(sequence): g = sequence.count("G") c = sequence.count("C") return round(((g + c) / len(sequence)) * 100, 2) def detect_mutations(reference_seq, sample_seq): alignments = pairwise2.align.globalms( reference_seq, sample_seq, 2, -1, -2, -0.5 ) best_alignment = alignments[0] aligned_ref = best_alignment.seqA aligned_sample = best_alignment.seqB mutations = [] alignment_visual = [] position = 0 for ref_base, sample_base in zip(aligned_ref, aligned_sample): # Skip double gaps if ref_base == "-" and sample_base == "-": continue # Track actual nucleotide position if ref_base != "-": position += 1 # Ignore insertion/deletion artifacts if ref_base == "-" or sample_base == "-": alignment_visual.append(" ") continue # Match if ref_base == sample_base: alignment_visual.append("|") # Real substitution mutation else: alignment_visual.append("*") mutations.append({ "position": position, "change": f"{ref_base}→{sample_base}", "probability": 95, "label": "HIGH", "effect": "Substitution", "context": "Actual mismatch detected", "position_percent": round( (position / len(reference_seq)) * 100, 2 ) }) alignment_text = ( aligned_ref + "\n" + "".join(alignment_visual) + "\n" + aligned_sample ) return mutations, alignment_text @app.route("/", methods=["GET", "POST"]) def home(): if request.method == "POST": try: sample_input = request.form["sample_sequence"] reference_input = request.form["reference_sequence"] sample_records = parse_fasta(sample_input) reference_records = parse_fasta(reference_input) if not sample_records or not reference_records: return render_template( "index.html", error="Invalid FASTA sequence." ) sample_record = sample_records[0] reference_record = reference_records[0] sample_seq = str(sample_record.seq).upper() reference_seq = str(reference_record.seq).upper() mutations, alignment = detect_mutations( reference_seq, sample_seq ) result = [{ "id": sample_record.id, "description": sample_record.description, "sequence": sample_seq, "length": len(sample_seq), "gc": calculate_gc(sample_seq), "mutation_count": len(mutations), "alignment": alignment, "top_mutations": mutations }] return render_template( "index.html", result=result ) except Exception as e: return render_template( "index.html", error=str(e) ) return render_template("index.html") if __name__ == "__main__": app.run(debug=True)
